   123→    lastName: '',
   124→    email: '',
   125→    phone: '',
   126→    nationality: '',
   127→    program: '',
   128→    branch: '',
   129→    duration: '',
   130→    startDate: '',
   131→    endDate: '',
   132→    message: '',
   133→    referral: '',

   135→    courseOfStudy: '',

   139→    airline: '',
   140→    arrivalDate: '',

   143→
   144→  // Calculate end date from start date + duration
   145→  const calculateEndDate = (startDate: string, duration: string): string => {
   146→    if (!startDate || !duration) return ''
   147→    const start = new Date(startDate)
   148→    const days = durationDaysMap[duration]
   149→    if (!days) return ''
   150→    const end = new Date(start)
   151→    end.setDate(end.getDate() + days)
   152→    return end.toISOString().split('T')[0]
   153→  }
   154→

   156→    if (!dateStr) return ''
   157→    const date = new Date(dateStr + 'T00:00:00')
   158→    return date.toLocaleDateString('en-US', { weekday: 'short', year: 'numeric', month: 'long', day: 'numeric' })
   159→  }
   160→
   161→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement>) => {
   162→    setFormData({ ...formData, [e.target.name]: e.target.value })
   163→  }
   164→

   166→  useEffect(() => {
   167→    const programParam = searchParams.get('program')
   168→    if (programParam && programs.includes(programParam)) {
   169→      setFormData(prev => ({ ...prev, program: programParam }))
   170→    }
   171→    const branchParam = searchParams.get('branch')
   172→    if (branchParam) {
   173→      setFormData(prev => ({ ...prev, branch: branchParam }))
   174→    }
   175→    const startDateParam = searchParams.get('startDate')
   176→    if (startDateParam) {
   177→      setFormData(prev => ({ ...prev, startDate: startDateParam }))
   178→    }
   179→    const durationParam = searchParams.get('duration')
   180→    if (durationParam && durations.includes(durationParam)) {
   181→      setFormData(prev => ({ ...prev, duration: durationParam }))
   182→    }
