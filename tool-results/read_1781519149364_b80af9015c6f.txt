   507→// Fuzzy match score - returns a score from 0 to 1
   508→function fuzzyMatch(query: string, text: string): number {
   509→  const q = query.toLowerCase().trim()
   510→  const t = text.toLowerCase().trim()
   511→
   512→  // Exact match
   513→  if (t === q) return 1.0
   514→  // Starts with
   515→  if (t.startsWith(q)) return 0.9
   516→  // Contains
   517→  if (t.includes(q)) return 0.8
   518→  // Word boundary match
   519→  const words = t.split(/\s+/)
   520→  if (words.some(w => w.startsWith(q))) return 0.7
   521→  // Any word contains
   522→  if (words.some(w => w.includes(q))) return 0.6
   523→
   524→  // Character-by-character fuzzy match
   525→  let qi = 0
   526→  let score = 0
   527→  let lastMatchIndex = -1
   528→
   529→  for (let ti = 0; ti < t.length && qi < q.length; ti++) {
   530→    if (t[ti] === q[qi]) {
   531→      score += 1
   532→      // Bonus for consecutive matches
   533→      if (lastMatchIndex === ti - 1) score += 0.5
   534→      lastMatchIndex = ti
   535→      qi++
   536→    }
   537→  }
   538→
   539→  // If we didn't match all query characters, it's not a match
   540→  if (qi < q.length) return 0
   541→
   542→  return (score / q.length) * 0.5
   543→}

   547→  score: number
   548→  matchReason: string

   551→export interface SearchFilters {
   552→  category?: SearchCategory
   553→  destination?: string
   554→  programType?: string
   555→  interests?: string[]
   556→}

   558→export function search(
   559→  query: string,
   560→  filters?: SearchFilters,
   561→  limit: number = 20
   562→): SearchResult[] {
   563→  if (!query.trim()) return []
   564→
   565→  const results: SearchResult[] = []
   566→
   567→  for (const item of searchIndex) {
   568→    // Apply filters first
   569→    if (filters?.category && item.category !== filters.category) continue
   570→    if (filters?.destination && item.destination !== filters.destination) continue
   571→    if (filters?.programType && item.programType !== filters.programType) continue
   572→
   573→    // Calculate match score
   574→    let bestScore = 0
   575→    let matchReason = ''
   576→
   577→    // Title match (highest weight)
   578→    const titleScore = fuzzyMatch(query, item.title)
   579→    if (titleScore > bestScore) {
   580→      bestScore = titleScore * 1.5
   581→      matchReason = 'title'
   582→    }
   583→
   584→    // Keywords match
   585→    for (const keyword of item.keywords) {
   586→      const kwScore = fuzzyMatch(query, keyword)
   587→      if (kwScore * 1.2 > bestScore) {
   588→        bestScore = kwScore * 1.2
   589→        matchReason = 'keyword'
   590→      }
   591→    }
   592→
   593→    // Description match (lower weight)
   594→    const descScore = fuzzyMatch(query, item.description)
   595→    if (descScore * 0.8 > bestScore) {
   596→      bestScore = descScore * 0.8
   597→      matchReason = 'description'
   598→    }
   599→
   600→    // Tags match
   601→    for (const tag of item.tags) {
   602→      const tagScore = fuzzyMatch(query, tag)
   603→      if (tagScore * 1.1 > bestScore) {
   604→        bestScore = tagScore * 1.1
   605→        matchReason = 'tag'
   606→      }
   607→    }
   608→
   609→    // Personalization boost based on interests
   610→    if (filters?.interests && item.relatedInterests) {
   611→      const overlap = filters.interests.filter(i => item.relatedInterests!.includes(i))
   612→      if (overlap.length > 0) {
   613→        bestScore += overlap.length * 0.15
   614→      }
   615→    }
   616→
   617→    if (bestScore >= 0.3) {
   618→      results.push({
   619→        item,
   620→        score: Math.min(bestScore, 1.0),
   621→        matchReason,
   622→      })
   623→    }
   624→  }
   625→
   626→  // Sort by score descending
   627→  results.sort((a, b) => b.score - a.score)
   628→
   629→  return results.slice(0, limit)
   630→}

   632→// Type-ahead suggestions - returns quick suggestions as user types
   633→export function getSuggestions(query: string, limit: number = 8): SearchResult[] {
   634→  if (!query.trim() || query.length < 2) return []
   635→
   636→  return search(query, undefined, limit)
   637→}

   658→      .filter(item => item.category === 'program')

   661→  }
   662→
   663→  const results: SearchResult[] = []
   664→  for (const item of searchIndex) {
   665→    if (item.relatedInterests) {
   666→      const overlap = interests.filter(i => item.relatedInterests!.includes(i))
   667→      if (overlap.length > 0) {
   668→        results.push({
   669→          item,
   670→          score: overlap.length * 0.3,
   671→          matchReason: 'personalized',
   672→        })
   673→      }
   674→    }
   675→  }
   676→
   677→  results.sort((a, b) => b.score - a.score)
   678→  return results.slice(0, 6)
