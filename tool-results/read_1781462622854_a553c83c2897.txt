   257→export default function OutreachDonationModal({
   258→  open,
   259→  onClose,
   260→  programArea,
   261→}: {
   262→  open: boolean
   263→  onClose: () => void
   264→  programArea: string
   265→}) {
   266→  const config = programConfigs[programArea]
   267→  const [submitted, setSubmitted] = useState(false)
   268→  const [submitting, setSubmitting] = useState(false)
   269→  const [formData, setFormData] = useState({
   270→    fullName: '',
   271→    email: '',
   272→    phone: '',
   273→    organization: '',
   274→    country: '',

   282→    message: '',
   283→  })
   284→
   285→  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
   286→    setFormData({ ...formData, [e.target.name]: e.target.value })
   287→  }
   288→
   289→  const handleSubmit = async (e: React.FormEvent) => {
   290→    e.preventDefault()
   291→    if (submitting) return
   292→    setSubmitting(true)
   293→    try {
   294→      const res = await fetch('/api/outreach-donations', {
   295→        method: 'POST',
   296→        headers: { 'Content-Type': 'application/json' },
   297→        body: JSON.stringify({ ...formData, programArea }),
   298→      })
   299→      if (res.ok) {
   300→        setSubmitted(true)
   301→      }
   302→    } catch (err) {
   303→      console.error('Outreach donation submission failed:', err)
   304→    } finally {
   305→      setSubmitting(false)
   306→    }
   307→  }
   308→
   309→  const handleClose = () => {
   310→    setSubmitted(false)
   311→    setFormData({
   312→      fullName: '',
   313→      email: '',
   314→      phone: '',
   315→      organization: '',
   316→      country: '',
   317→      city: '',
   318→      category: '',
   319→      itemCondition: '',
   320→      quantity: '',
   321→      description: '',
   322→      deliveryMethod: '',
   323→      pickupAddress: '',
   324→      message: '',
   325→    })
   326→    onClose()
   327→  }
   328→
   329→  if (!config) return null
   330→
   331→  const ConfigIcon = config.icon
   332→
   333→  if (submitted) {
   334→    return (
   335→      <Dialog open={open} onOpenChange={handleClose}>

   343→            >
   344→              <div className="w-20 h-20 bg-vogue/10 rounded-full flex items-center justify-center mx-auto mb-6">
   345→                <CheckCircle2 className="w-10 h-10 text-vogue" />
   346→              </div>
   347→              <h2 className="text-2xl font-bold text-cornell mb-3">Thank You!</h2>
   348→              <p className="text-charcoal dark:text-white/70 mb-2">
   349→                Your donation offer for <span className="font-semibold text-cornell">{programArea}</span> has been received.
   350→              </p>
   351→              <p className="text-sm text-charcoal/70 dark:text-white/50 mb-6">
   352→                Our team will review your submission and contact you at <span className="font-semibold">{formData.email}</span> within 48 hours to coordinate the donation and delivery details.
   353→              </p>
   354→              <Button
   355→                onClick={handleClose}
   356→                className="bg-cornell hover:bg-cornell-dark text-white rounded-full px-8"
   357→              >
   358→                Close
   359→              </Button>
   360→            </motion.div>
   361→          </div>
   362→        </DialogContent>
   363→      </Dialog>
   364→    )
   365→  }
   366→
   367→  return (
   368→    <Dialog open={open} onOpenChange={handleClose}>
   369→      <DialogContent className="sm:max-w-2xl bg-white dark:bg-[#0A1F12] p-0 overflow-hidden max-h-[90vh] overflow-y-auto">
   370→        <DialogTitle className="sr-only">Donate - {config.title}</DialogTitle>
   371→
   372→        {/* Header */}
   373→        <div className={`bg-gradient-to-br ${config.color} p-6 text-white relative`}>
   374→          <button
   375→            onClick={handleClose}

   387→            </div>

   413→                  className="w-full h-10 px-3 rounded-lg border border-border bg-cream dark:bg-[#122A1B] text-charcoal dark:text-white text-sm focus:border-cornell focus:ring-cornell focus:outline-none"
